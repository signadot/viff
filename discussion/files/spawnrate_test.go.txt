// git-issue bteb1a97h12kr7nagnn0: the experiment the issue names — the same fan-out with a
// spawn and with a plain patch.
func TestZZSpawnRate(t *testing.T) {
	for _, tc := range []struct {
		name, action string
		n            int
		wait         time.Duration
	}{
		{"instant", spawnAction, 20, 0},
		{"instant", spawnAction, 60, 0},
		{"200ms", spawnAction, 20, 200 * time.Millisecond},
		{"200ms", spawnAction, 60, 200 * time.Millisecond},
	} {
		n := tc.n
		taskTime = tc.wait
		t.Run(fmt.Sprintf("%s-%d", tc.name, n), func(t *testing.T) {
			store := entitytest.Store(t)
			eng := New(store, task.AutoResolver{})
			eng.SetSpawner(func(id string, spec launch.Spec) task.Handle {
				sup := task.NewSupervisor(&task.Vector{Default: task.Auto}, task.AutoResolver{})
				p := task.NewTask(id, sup, slowDriver{d: taskTime}, noExec{})
				mirror.Reflect(store, id, p, mirror.Meta{
					Subject: spec.Subject, Runner: "local", Chain: spec.Chain,
					Prompt: spec.Prompt, Occurrence: spec.Occurrence,
				})
				go p.Run(context.Background())
				return p
			})
			if err := eng.AddTrigger(mustParseCharter(t, `
- name: r
  condition: {subject: {system: demo, kind: pr, value: {stage: open}}}
  action:
    `+tc.action+`
`)); err != nil {
				t.Fatal(err)
			}
			ctx, cancel := context.WithCancel(context.Background())
			defer cancel()
			eng.ObserveScoped(ctx)

			rev0 := store.Head()
			start := time.Now()
			for i := 0; i < n; i++ {
				entitytest.Set(t, store, prRef(fmt.Sprint(i)), entity.Obj{"stage": "open"}.Payload())
			}
			landed := time.Since(start)

			deadline := time.After(180 * time.Second)
			done := 0
			for done < n {
				done = 0
				for i := 0; i < n; i++ {
					e, _, _ := store.Get(prRef(fmt.Sprint(i)))
					if entity.Str(e.Payload, "stage") == "done" {
						done++
					}
				}
				if done >= n {
					break
				}
				select {
				case <-deadline:
					t.Fatalf("%s: only %d/%d after %v", tc.name, done, n, time.Since(start))
				case <-time.After(20 * time.Millisecond):
				}
			}
			total := time.Since(start)
			byKind := map[string]int{}
			for _, e := range store.List(nil) {
				byKind[e.Ref.Root()+"/"+e.Ref.System()+":"+e.Ref.Kind()]++
			}
			t.Logf("  entities: %v", byKind)
			t.Logf("%-11s %d subjects: landed in %v, all done in %v (%.1f/s), %d commits (%.1f per subject)",
				tc.name, n, landed.Round(time.Millisecond), total.Round(time.Millisecond),
				float64(n)/total.Seconds(), store.Head()-rev0,
				float64(store.Head()-rev0)/float64(n))
		})
	}
}

const spawnAction = "patch: {stage: claimed}\n    spawn: {prompt: \"go\"}\n    onDone: {patch: {stage: done}}"

var taskTime time.Duration

// slowDriver takes a realistic moment to finish, so the engine's live-task dedupe has the same
// window it has in production.
type slowDriver struct{ d time.Duration }

func (s slowDriver) Next(ctx context.Context, _ task.StepResult) (task.ToolUse, bool, error) {
	select {
	case <-time.After(s.d):
	case <-ctx.Done():
	}
	return task.ToolUse{}, false, nil
}
